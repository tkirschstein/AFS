	\begin{algorithmic}[1]
		\Procedure{TourDest}{$T$, $rule^{del}$}
			\State randomly select a number $q$ of requests to be deleted: $q \in \left\lbrace 1,...,|\Rc^T|-1\right\rbrace$ 
			\For{i = 1,...,q}
				\State $T \leftarrow \textsc{DelReq}(T, rule^{del})$
			\EndFor
			\State 	\Return $T$
			
		\EndProcedure
	\end{algorithmic}
