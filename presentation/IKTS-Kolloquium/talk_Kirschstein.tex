& t_{lf} \cdot P^{UAV}\left(m,\nu,0^\circ\right) + & \text{level flight}\\
& \left.  t_{hover} \cdot  P^{UAV}\left(m,|\nu_{head}|,0^\circ\right) \right) + & \text{hovering}\\
& t \cdot \frac{P^{aux}}{\epsilon_{UAV}^{charg}} & \text{auxiliaries} 
\end{align*}
\vfill
$\Longrightarrow$ $t=t_{lf} + t_{hover} + 2 \cdot t_{tol}$ with $t_{lf} = \frac{d}{v_{net}} - 2 \cdot t_{tol}$ and $t_{tol} = \frac{a}{v_{net}}$
\vfill
