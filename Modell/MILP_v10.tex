\section{1. Index Sets}\label{index-sets}

\begin{align}
i \in \mathcal{I} &= \{1, \ldots, n_s\} && \text{Agroforestry sites} \\
j \in \mathcal{J} &= \{1, \ldots, n_j\} && \text{Storage/processing facilities} \\
k \in \mathcal{K} &= \{1, \ldots, n_k\} && \text{Consumer sites} \\
t \in \mathcal{T} &= \{1, \ldots, T_{\max}\} && \text{Time periods (years)} \\
t \in \mathcal{T^+} &= \{0, \ldots, T_{\max}+1\} && \text{Time periods (years)} \\
t \in \mathcal{T^{harv}} &= \{A^{min}+1, \ldots, T_{\max}\} && \text{ Harvesting periods (years)} \\
p \in \mathcal{P} &= \{1, 2, 3\} && \text{Product types (chemical, pulp, energy)} \\
(s,t) \in \mathcal{S} &= \left\{(s,t) | s,t \in 0,...,(T_{\max}+1) \wedge (t-s) \in \left\{A^{min},...,A^{max}\right\} \right\} && \text{Consecutive harvesting}
\end{align}

If establishment occurs in period 3 and harvests in periods 6 and 10,
the following variables would be active:

\begin{itemize}
\tightlist
\item
  \(z_{i,0,3}=1\) (establishment),
\item
  \(z_{i,3,6}=1\) (harvest in 6 with establishment in 3),
\item
  \(z_{i,6,10}=1\) (harvest in 10 with predecessor in 6),
\item
  \(z_{i,10,11}=1\) (last harvest in 10),and
\item
  \(z_{ist}=0\) for all \((s,t) \in \mathcal{S}\)
\end{itemize}

Constraints C1 and C2 read like this

\begin{itemize}
\tightlist
\item
  C1: \(\sum_{t=1}^{10} z_{i,0,t} \leq 1\) (at most one establishment)
  \(\rightarrow\) as \(z_{i,0,3}=1\) it follows that \(z_{i,0,t}=0\) for
  all \(t \neq 3\).
\item
  C2\_1: \(z_{i,0,1} = \sum_{u \in 2,...,6,11} z_{i,1,u}\) (if there is
  establishment in 1, there must be an harvest in \(u\) or closing)
  \(\rightarrow\) as \(z_{i,0,1}=0\),
  \(\sum_{u\in 2,...,6,11} z_{i,1,u} = 0\).
\item
  C2\_2: \(z_{i,0,2} + z_{i,1,2} = \sum_{u \in 3,...,7,11} z_{i,2,u}\)
  (if there is establishment or harvest in 2, there must be an harvest
  in \(u\) or closing) \(\rightarrow\) as \(z_{i,0,2}=z_{i,1,2}=0\),
  \(\sum_{u\in 3,...,7,11} z_{i,3,u} = 0\)
\item
  C2\_3:
  \(z_{i,0,3} + z_{i,1,3} + z_{i,2,3} = \sum_{u \in 4,...,8,11} z_{i,3,u}\)
  (if there is establishment or harvest in 3, there must be an harvest
  in \(u\) or closing) \(\rightarrow\) as \(z_{i,0,3}=1\) and
  \(z_{i,1,3}=z_{i,2,3}=0\), \(\sum_{u\in 4,...,8,11} z_{i,3,u} = 1\)
  \(\rightarrow\) as \(z_{i,3,6}=1\), it follows that \(z_{i,3,u}=0\)
  for all \(u \neq 6\).
\item
  C2\_4:
  \(z_{i,0,4} + z_{i,1,4} + z_{i,2,4} + z_{i,3,4} = \sum_{u \in 5,...,9,11} z_{i,4,u}\)
  (if there is establishment or harvest in 4, there must be an harvest
  in \(u\) or closing) \(\rightarrow\) as
  \(z_{i,0,4} + z_{i,1,4} + z_{i,2,4} + z_{i,3,4} =0\), it follows that
  \(\sum_{u \in 5,...,9,11} z_{i,4,u}=0\).
\item
  and so on opjas
\end{itemize}

\section{2. Decision Variables}\label{decision-variables}

\subsection{2.1 Binary Variables}\label{binary-variables}

\begin{align}
z_{ist} & \in \{0, 1\} && \text{Indicator if at site } i \text{ in periods} t,s \text{ are consecutive harvests with } (s,t) \in \mathcal{S}
\end{align}

\subsection{2.2 Continuous Variables}\label{continuous-variables}

\begin{align}
X_{ijpt} &\geq 0 && \text{Flow of product } p \in \mathcal{P} \text{ from site } i \in \mathcal{I}  \to \text{ storage } j \in \mathcal{J} \text{ in period } t \in \mathcal{T^{harv}} \text{ [tonnes]} \\
X_{jkpp't} &\geq 0 && \text{Conversion flow of product } p \in \mathcal{P}\text{ to product } p'\in \mathcal{P}: p'\geq p \text{ from storage } j \to \text{ consumer } k \text{ in period } t  \in \mathcal{T^{harv}} \text{ [tonnes]}   \\
Y_{ipt} &\geq 0 && \text{Maximum harvest quantity of product } p \text{ at site } i  \text{ in period } t  \in \mathcal{T^{harv}} \text{ [tonnes]} \\
S_{jpt} &\geq 0 && \text{Inventory at storage } j \text{ in period } t  \in \mathcal{T}\text{ [tonnes]}
\end{align}

\section{4. Objective Function}\label{objective-function}

\begin{align}
\max \, Z = \quad
&\sum_{k,p,t} R_{k,p} \cdot \sum_{j \in \mathcal{J}, p'\in \mathcal{P}: p' \leq p } X_{jkp'pt} && \text{Revenue} \\
&- \sum_{i} C^{est}_i \cdot \text{AREA}_i \cdot \sum_{t \in T} z_{i0t} && \text{Establishment} \\
&- \sum_{i} C^{opp}_i \cdot \text{AREA}_i \cdot \sum_{t \in T} (T^{max}-t) \cdot z_{i0t} && \text{Opportunity} \\
&- \sum_{i,(s,t)\in \mathcal{S}| s>0} C^{harv}_i \cdot \text{AREA}_i \cdot z_{ist} && \text{Harvest} \\
&- c^{tr-raw}\cdot\sum_{i,j,p,t} d_{ij} \cdot X_{ijpt} && \text{Transport (site→hub)} \\
&- c^{tr-pre}\cdot \sum_{j,k,p,t} d_{jk} \cdot X_{jkpt} && \text{Transport (hub→consumer)} \\
&- \sum_{j,p,t} c^{stor}_j \cdot S_{jpt} && \text{Storage}
\end{align}

\section{5. Constraints}\label{constraints}

\subsection{Constraint Set 1: AFS
establishment}\label{constraint-set-1-afs-establishment}

\begin{align}
\sum_{t \in \mathcal{T^+}} z_{i0t} &\leq 1 & \forall i \in \mathcal{I} \tag{C1}
\end{align}

\textbf{Interpretation:} If AFS is established in \(t\),
\(z_{i,0,t}=1\).

\begin{center}\rule{0.5\linewidth}{0.5pt}\end{center}

\subsection{Constraint Set 2: Path
connectivity}\label{constraint-set-2-path-connectivity}

\begin{align}
\sum_{ (s,t) \in \mathcal{S}} z_{ist} &= \sum_{(t,u) \in \mathcal{S}} z_{itu} & \forall i \in \mathcal{I}, t \in \mathcal{T} \tag{C2}
\end{align}

\textbf{Interpretation:} If there is a harvest in \(t\), there must be
an predecessor in \(s<t\) and successor in \(u>t\).

\begin{center}\rule{0.5\linewidth}{0.5pt}\end{center}

\subsection{Constraint Set 3: Biomass yield
calculation}\label{constraint-set-3-biomass-yield-calculation}

\begin{align}
Y_{ipt} &= \sum_{s=1}^{t-1} \eta_{p(t-s)} \cdot AREA_i \cdot z_{ist}  & \forall i \in \mathcal{I}, p \in \mathcal{P}, t \in \mathcal{T^{harv}} \tag{C3} \\
\end{align}

\textbf{Interpretation:} Biomass yield is restricted by age of plantage

\begin{center}\rule{0.5\linewidth}{0.5pt}\end{center}

\subsection{Constraint Set 4: Yield Shipping
Linkage}\label{constraint-set-4-yield-shipping-linkage}

\begin{align}
\sum_{j \in \mathcal{J}} X_{ijpt} &\leq Y_{ipt} & \forall i \in \mathcal{I}, p \in \mathcal{P}, t \in \mathcal{T^{harv}} \tag{C4}
\end{align}

\textbf{Interpretation:} Biomass shipped from site \(i\) (product \(p\),
period \(t\)) is limited by available harvest.

\begin{center}\rule{0.5\linewidth}{0.5pt}\end{center}

\subsection{Constraint Set 5: Inventory
Balance}\label{constraint-set-5-inventory-balance}

\begin{align}
S_{jpt} &= S_{jp(t-1)} + \sum_{i} X_{ijpt} - \sum_{k, p' \geq p} X_{jkpp't} & \forall j \in \mathcal{J}, p \in \mathcal{P}, t \in \mathcal{T^{harv}} \tag{C5}
\end{align}

with \(S_{j,p,0} = 0\).

\textbf{Interpretation:} Flow conservation at storage hubs.

\begin{center}\rule{0.5\linewidth}{0.5pt}\end{center}

\subsection{Constraint Set 6: Storage
Capacity}\label{constraint-set-6-storage-capacity}

\begin{align}
\sum_{p \in \mathcal{P}} S_{jpt} &\leq \text{CAP}^{stor}_j & \forall j \in\mathcal{J}, t \in \mathcal{T^{harv}} \tag{C6}
\end{align}

\textbf{Interpretation:} Total inventory limited by physical storage
space.

\begin{center}\rule{0.5\linewidth}{0.5pt}\end{center}

\subsection{Constraint Set 7: Processing
Capacity}\label{constraint-set-7-processing-capacity}

\begin{align}
\sum_{i \in \mathcal{I}, p \in \mathcal{P}} X_{ijpt} &\leq \text{CAP}^{proc}_j & \forall j \in \mathcal{J}, t \in \mathcal{T^{harv}} \tag{C7}
\end{align}

\textbf{Interpretation:} Incoming material throughput limited by
processing equipment.

\begin{center}\rule{0.5\linewidth}{0.5pt}\end{center}

\subsection{Constraint Set 8: Demand Satisfaction with Product
Cascade}\label{constraint-set-8-demand-satisfaction-with-product-cascade}

\begin{align}
\sum_{j \in \mathcal{J}, p'\in \mathcal{P}: p' \leq p } X_{jkp'pt} & \leq D^{max}_{kpt} & \forall k \in  \mathcal{K}, p \in  \mathcal{P}, t \in \mathcal{T^{harv}} \tag{C8}
\end{align}

\textbf{Interpretation:} Higher-quality products (lower \(p\)) can
substitute for lower-quality demand.

\textbf{Cascade hierarchy:} - Product 1 (chemical) → Satisfies any
demand - Product 2 (pulp) → Satisfies product 2 or 3 demand - Product 3
(energy) → Only satisfies product 3 demand
